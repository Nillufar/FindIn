CREATE TABLE USER (
	UserId Int NOT NULL IDENTITY (100, 2),
	SkillNumber Int NOT NULL UNIQUE,
	Projects Int NOT NULL UNIQUE,
	
	CONSTRAINTS USERID_PK PRIMARY KEY(UserId, SkillNumber, Projects),
	CONSTRAINTS SKILLS_NUMBER_FK FOREIGN KEY(SkillNumber)
		REFERENCES SKILLS(SkillId),
	CONSTRAINTS PROJECTS_FK FOREIGN KEY(Projects)
		REFERENCES PROJECTS(ProjectId)
);

CREATE TABLE SKILLS (
	SkillId Int NOT NULL IDENTITY (5000, 100),
	UserNumber Int NOT NULL UNIQUE,
	ProjectNumber Int NOT NULL UNIQUE,
	Skills VarChar (100) NULL,

	CONSTRAINTS SKILLID_PK PRIMARY KEY(SkillId, UserNumber),
	CONSTRAINTS USERID_FK FOREIGN KEY(UserNumber)
		REFERENCES USER(UserId),
	CONSTRAINST PROJETID_FK FOREIGN KEY(ProjectNumber)
		REFERENCES PROJECTS_FK FOREIGN KEY(ProjectNumber)
);

CREATE TABLE PROJECTS(
	ProjectId Int NOT NULL IDENTITY (1000, 55),
	UserNumber Int NOT NULL UNIQUE,

	CONSTRAINTS PROJECT_ID_PK PRIMARY KEY(ProjectId, UserNumber),
	CONSTRAUNTS USER_NUMBER_FK FOREIGN KEY(UserNumber)
		REFERENCE USER(UserId)
);


CREATE TABLE POPULATION (
	UserNumber Int NOT NULL UNIQUE,
	FirstName Char(25) NOT NULL,
	LastName Char(50) NOT NULL,
	Email VarChar (100) NOT NULL UNIQUE,
	Skills  Int NOT NULL UNIQUE,
	Projects Int NOT NULL UNIQUE,
	
	CONSTRAINTS INDEVIDUAL_PK PRIMARY KEY(UserNumber, FirstName, LastName, Email, Skills, Projects),
	CONSTRAINTS USER_NUMBER_FK FOREIGN KEY(UserNumber)
		REFERENCES USER(UserId),
	CONSTRAINTS PROJECTS_FK FOREIGN KEY(ProjectNumber)
		REFERENCES PROJECTS(ProjectId),
	CONSTRAINTS SKILLSID_FK FOREIGN KEY(Skills)
		REFERENCES SKILLS(SkillId)
	
);


CREATE TABLE WORK (
	UserNumber Int NOT NULL UNIQUE,
	SkillNumber Int NOT NULL UNIQUE,
	ProjectNumber Int NOT NULL UNIQUE,
	CurrentProject VarChar(200) NULL DEFAULT 'Not currently on a project.',
	PastProjects VarChar(200) NULL DEFAULT 'Have not done a project yet.',

	CONSTRAINTS WORK_HISTORY_PK PRIMARY KEY(UserNumber, SkillNumber, ProjectNumber, CurrentProject, PastProjects),
	CONSTRAINTS USER_NUMBER_FK FOREIGN KEY(UserNumber)
		REFERENCE USER(UserId),
	CONSTRAINTS PROJECT_NUMBER_FK FOREIGN KEY(ProjectNumber)
		REFERENCE PROJECTS(ProjectId),
	CONSTRAINTS SKILL_NUMBER_FK FOREIGN KEY(SkillNumber)
		REFERENCE SKILLS(SkillId)
);

CREATE PROCEDURE OR REPLACE PROCEDURE AddDynamicColumn
AS
BEGIN
	DECLARE @ColumnName := 'Skills', 
		@Apend VarChar(100);
	SELECT @ColumnName = Skills
	FROM SKILLS
	

	IF EXISTS (
		SELECT 1
		FROM INFORMATION_SCHEMA.COLUMNS
		WHERE TABLE_NAME = 'SKILLS' AND COLUMN_NAME = @ColumnName
	)
	BEGIN
		UPDATE SKILLS
		SET @Apend = QUOTENAME(@ColumnName) + ' VarChar(10)';
		
		EXEC AddDynamicColumn(@Apend);
	END
END;
